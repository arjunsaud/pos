   126→/* ------------------------------------------------------------------ */
   127→/*  Helper: Render markdown-like content from store                     */
   128→/* ------------------------------------------------------------------ */
   129→function RenderMarkdown({ content }: { content: string }) {
   130→  const lines = content.split('\n');
   131→  const elements: React.ReactNode[] = [];
   132→  let key = 0;
   133→
   134→  for (const line of lines) {
   135→    const trimmed = line.trim();
   136→    if (!trimmed) {
   137→      elements.push(<div key={key++} className="h-2" />);
   138→      continue;
   139→    }
   140→    if (trimmed.startsWith('### ')) {
   141→      elements.push(<h3 key={key++} className="text-base font-semibold mt-4">{trimmed.slice(4)}</h3>);
   142→    } else if (trimmed.startsWith('## ')) {
   143→      elements.push(<h2 key={key++} className="text-lg font-semibold mt-4">{trimmed.slice(3)}</h2>);
   144→    } else if (trimmed.startsWith('# ')) {
   145→      elements.push(<h1 key={key++} className="text-xl font-bold mt-2">{trimmed.slice(2)}</h1>);
   146→    } else if (trimmed.startsWith('- ')) {
   147→      const boldMatch = trimmed.match(/^- (\*\*(.+?)\*\*:)?(.*)$/);
   148→      const label = boldMatch?.[2];
   149→      const rest = boldMatch?.[3] || trimmed.slice(2);
   150→      elements.push(
   151→        <li key={key++} className="text-sm text-muted-foreground ml-4 list-disc">
   152→          {label && <strong className="text-foreground font-medium">{label}:</strong>}
   153→          <span dangerouslySetInnerHTML={{ __html: rest.replace(/\*\*(.+?)\*\*/g, '<strong class="font-medium text-foreground">$1</strong>') }} />
   154→        </li>
   155→      );
   156→    } else {
   157→      elements.push(
   158→        <p key={key++} className="text-sm text-muted-foreground leading-relaxed" dangerouslySetInnerHTML={{ __html: trimmed.replace(/\*\*(.+?)\*\*/g, '<strong class="font-medium text-foreground">$1</strong>') }} />
   159→      );
   160→    }
   161→  }
   162→
   163→  return <div className="space-y-1">{elements}</div>;
   164→}
