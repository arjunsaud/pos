    65→    65→function generateSpendingTrend(totalSpent: number) {
    66→    66→  const now = new Date();
    67→    67→  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
    68→    68→  const base = totalSpent / 6;
    69→    69→  return Array.from({ length: 6 }, (_, i) => {
    70→    70→    const d = new Date(now.getFullYear(), now.getMonth() - 5 + i, 1);
    71→    71→    const factor = 0.5 + Math.random() * 1.0;
    72→    72→    return {
    73→    73→      month: months[d.getMonth()],
    74→    74→      amount: Math.round(base * factor),
    75→    75→    };
    76→    76→  });
    77→    77→}

    79→    79→function simpleHash(str: string): number {
    80→    80→  let hash = 0;
    81→    81→  for (let i = 0; i < str.length; i++) {
    82→    82→    hash = ((hash << 5) - hash) + str.charCodeAt(i);
    83→    83→    hash |= 0;
    84→    84→  }
    85→    85→  return Math.abs(hash);
    86→    86→}

    88→    88→function generateMockOrders(customer: Customer) {
    89→    89→  const statuses = ORDER_STATUSES;
    90→    90→  const seed = simpleHash(customer.id);
    91→    91→  const baseDate = new Date(customer.lastVisit);
    92→    92→  return Array.from({ length: 5 }, (_, i) => {
    93→    93→    const date = new Date(baseDate);
    94→    94→    date.setDate(date.getDate() - (i + 1) * Math.floor(3 + (seed % 10)));
    95→    95→    const itemCount = 1 + ((seed * (i + 1)) % 5);
    96→    96→    const total = Math.round((customer.totalSpent / Math.max(customer.totalPurchases, 1)) * (0.5 + ((seed * (i + 2)) % 10) / 10) * 100) / 100;
    97→    97→    const payment = PAYMENT_METHODS[(seed + i) % PAYMENT_METHODS.length];
    98→    98→    const status = customer.totalPurchases > 0 ? statuses[i] : 'pending';
    99→    99→    const invNum = `INV-${String(1000 + ((seed + i * 7) % 9000))}`;
   100→   100→    return { invoiceNumber: invNum, date: date.toISOString(), items: itemCount, total, paymentMethod: payment, status };
   101→   101→  });
   102→   102→}
