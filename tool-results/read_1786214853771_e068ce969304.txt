    89→function formatDate(date: Date): string {
    90→  const y = date.getFullYear();
    91→  const m = String(date.getMonth() + 1).padStart(2, '0');
    92→  const d = String(date.getDate()).padStart(2, '0');
    93→  return `${y}-${m}-${d}`;
    94→}
