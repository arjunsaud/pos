     1→'use client';
     2→
     3→import { useState, useMemo } from 'react';
     4→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     5→import { Input } from '@/components/ui/input';
     6→import { Button } from '@/components/ui/button';
     7→import { Badge } from '@/components/ui/badge';
     8→import { ScrollArea } from '@/components/ui/scroll-area';
     9→import { Separator } from '@/components/ui/separator';
    10→import {
    11→  Dialog, DialogContent, DialogHeader, DialogTitle,
    12→} from '@/components/ui/dialog';
    13→import { Search, Plus, Minus, X, ShoppingCart, Banknote, CreditCard, Smartphone, Wallet, Receipt, Barcode, Trash2, TrendingUp, Clock, Package, Pause, PlayCircle } from 'lucide-react';
    14→import { mockProducts, mockCategories } from '@/lib/mock-data';
    15→import { toast } from 'sonner';
    16→import { PageHeader } from '@/components/shared/page-header';
    17→import type { CartItem, Product, HeldSale } from '@/lib/types';
    18→import { cn } from '@/lib/utils';
    19→import { formatDateTime, npr } from '@/lib/helpers';
    20→
    21→const PAYMENT_METHODS = [
    22→  { id: 'cash', label: 'Cash', icon: Banknote, color: 'text-emerald-600 dark:text-emerald-400' },
    23→  { id: 'card', label: 'Card', icon: CreditCard, color: 'text-blue-600 dark:text-blue-400' },
    24→  { id: 'esewa', label: 'eSewa', icon: Smartphone, color: 'text-green-600 dark:text-green-400' },
    25→  { id: 'khalti', label: 'Khalti', icon: Wallet, color: 'text-purple-600 dark:text-purple-400' },
    26→] as const;
    27→
    28→type PaymentMethodId = (typeof PAYMENT_METHODS)[number]['id'];
    29→
    30→export default function POSTerminal() {
    31→  const [searchQuery, setSearchQuery] = useState('');
    32→  const [barcodeInput, setBarcodeInput] = useState('');
    33→  const [selectedCategory, setSelectedCategory] = useState('All');
    34→  const [cart, setCart] = useState<CartItem[]>([]);
    35→  const [discount, setDiscount] = useState(0);
    36→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethodId>('cash');
    37→  const [receiptOpen, setReceiptOpen] = useState(false);
    38→  const [lastSale, setLastSale] = useState<{ items: CartItem[]; subtotal: number; discount: number; vat: number; total: number; method: string; time: string } | null>(null);
    39→  // Hold/Resume state
    40→  const [heldSales, setHeldSales] = useState<HeldSale[]>([]);
    41→  const [heldListOpen, setHeldListOpen] = useState(false);
    42→
    43→  const filteredProducts = useMemo(() => {
    44→    return mockProducts.filter((p) => {
    45→      if (!p.isActive) return false;
    46→      const q = searchQuery.toLowerCase();
    47→      const matchesSearch = !q || p.name.toLowerCase().includes(q) || p.sku.toLowerCase().includes(q);
    48→      const matchesCategory = selectedCategory === 'All' || p.category === selectedCategory;
    49→      return matchesSearch && matchesCategory;
    50→    });
    51→  }, [searchQuery, selectedCategory]);
    52→
    53→  const categories = ['All', ...mockCategories.map((c) => c.name)];
    54→
    55→  const addToCart = (product: Product) => {
    56→    if (product.stock <= 0) return;
    57→    setCart((prev) => {
    58→      const existing = prev.find((item) => item.product.id === product.id);
    59→      if (existing) {
    60→        if (existing.quantity >= product.stock) {
    61→          toast.error(`Maximum stock reached for ${product.name}`);
    62→          return prev;
    63→        }
    64→        return prev.map((item) =>
    65→          item.product.id === product.id
    66→            ? { ...item, quantity: item.quantity + 1 }
    67→            : item
    68→        );
    69→      }
    70→      return [...prev, { product, quantity: 1, discount: 0 }];
    71→    });
    72→  };
    73→
    74→  const updateQuantity = (productId: string, delta: number) => {
    75→    setCart((prev) =>
    76→      prev
    77→        .map((item) => {
    78→          if (item.product.id !== productId) return item;
    79→          const newQty = item.quantity + delta;
    80→          if (newQty <= 0) return null;
    81→          if (newQty > item.product.stock) {
    82→            toast.error(`Maximum stock reached for ${item.product.name}`);
    83→            return item;
    84→          }
    85→          return { ...item, quantity: newQty };
    86→        })
    87→        .filter(Boolean) as CartItem[]
    88→    );
    89→  };
    90→
    91→  const removeFromCart = (productId: string) => {
    92→    setCart((prev) => prev.filter((item) => item.product.id !== productId));
    93→  };
    94→
    95→  const subtotal = cart.reduce((sum, item) => sum + item.product.price * item.quantity, 0);
    96→  const discountAmount = Math.min(discount, subtotal);
    97→  const taxableAmount = subtotal - discountAmount;
    98→  const vat = taxableAmount * 0.13;
    99→  const total = taxableAmount + vat;
   100→
   101→  const handleBarcodeScan = () => {
   102→    if (!barcodeInput.trim()) return;
   103→    const product = mockProducts.find(p => p.sku.toLowerCase() === barcodeInput.trim().toLowerCase());
   104→    if (product) {
   105→      addToCart(product);
   106→      toast.success(`Added ${product.name} to cart`);
   107→    } else {
   108→      toast.error(`Product not found: ${barcodeInput}`);
   109→    }
   110→    setBarcodeInput('');
   111→  };
   112→
   113→  const handleHoldSale = () => {
   114→    if (cart.length === 0) {
   115→      toast.error('Cart is empty');
   116→      return;
   117→    }
   118→    const held: HeldSale = {
   119→      id: `held-${Date.now()}`,
   120→      cart: [...cart],
   121→      customerName: 'Walk-in',
   122→      heldAt: new Date().toISOString(),
   123→      total: Math.round(total * 100) / 100,
   124→    };
   125→    setHeldSales(prev => [held, ...prev]);
   126→    setCart([]);
   127→    setDiscount(0);
   128→    toast.success('Sale held! You can resume it later.');
   129→  };
   130→
   131→  const handleResumeSale = (held: HeldSale) => {
   132→    if (cart.length > 0) {
   133→      // Auto-hold current cart if not empty
   134→      const autoHeld: HeldSale = {
   135→        id: `held-${Date.now()}`,
   136→        cart: [...cart],
   137→        customerName: 'Walk-in',
   138→        heldAt: new Date().toISOString(),
   139→        total: Math.round(total * 100) / 100,
   140→      };
   141→      setHeldSales(prev => [autoHeld, ...prev]);
   142→    }
   143→    setCart(held.cart);
   144→    setHeldSales(prev => prev.filter(h => h.id !== held.id));
   145→    setHeldListOpen(false);
   146→    toast.success('Sale resumed!');
   147→  };
   148→
   149→  const handleDeleteHeld = (id: string) => {
   150→    setHeldSales(prev => prev.filter(h => h.id !== id));
   151→    toast.success('Held sale discarded');
   152→  };
   153→
   154→  const handleCompleteSale = () => {
   155→    if (cart.length === 0) {
   156→      toast.error('Cart is empty');
   157→      return;
   158→    }
   159→    const method = PAYMENT_METHODS.find((m) => m.id === paymentMethod);
   160→    const saleData = {
   161→      items: [...cart],
   162→      subtotal,
   163→      discount: discountAmount,
   164→      vat: Math.round(vat * 100) / 100,
   165→      total: Math.round(total * 100) / 100,
   166→      method: method?.label || 'Cash',
   167→      time: new Date().toISOString(),
   168→    };
   169→    setLastSale(saleData);
   170→    setReceiptOpen(true);
   171→    toast.success(`Sale completed! NPR ${npr(saleData.total)} via ${saleData.method}`);
   172→    setCart([]);
   173→    setDiscount(0);
   174→  };
   175→
   176→  return (
   177→    <div className="space-y-4">
   178→      <PageHeader
   179→        title="POS Terminal"
   180→        description="Process sales and accept payments"
   181→      >
   182→        <div className="flex items-center gap-2">
   183→          {heldSales.length > 0 && (
   184→            <Button variant="outline" size="sm" className="hidden sm:flex gap-1.5 border-amber-300 text-amber-700 dark:border-amber-700 dark:text-amber-400 hover:bg-amber-50 dark:hover:bg-amber-950/30" onClick={() => setHeldListOpen(true)}>
   185→              <Pause className="h-3.5 w-3.5" />
   186→              {heldSales.length} Held
   187→            </Button>
   188→          )}
   189→          <Badge variant="outline" className="hidden sm:flex gap-1.5">
   190→            <ShoppingCart className="h-3 w-3" />
   191→            {cart.length} items · NPR {npr(total)}
   192→          </Badge>
   193→        </div>
   194→      </PageHeader>
   195→
   196→      {/* Quick Stats Bar */}
   197→      <div className="grid grid-cols-3 gap-3">
   198→        <div className="flex items-center gap-2.5 rounded-xl border bg-card p-3 transition-shadow hover:shadow-md">
   199→          <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/30">
   200→            <TrendingUp className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   201→          </div>
   202→          <div className="min-w-0">
   203→            <p className="text-[11px] text-muted-foreground">Today's Sales</p>
   204→            <p className="text-sm font-bold">NPR {npr(35200)}</p>
   205→          </div>
   206→        </div>
   207→        <div className="flex items-center gap-2.5 rounded-xl border bg-card p-3 transition-shadow hover:shadow-md">
   208→          <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-blue-100 dark:bg-blue-900/30">
   209→            <ShoppingCart className="h-4 w-4 text-blue-600 dark:text-blue-400" />
   210→          </div>
   211→          <div className="min-w-0">
   212→            <p className="text-[11px] text-muted-foreground">Items Sold</p>
   213→            <p className="text-sm font-bold">47</p>
   214→          </div>
   215→        </div>
   216→        <div className="flex items-center gap-2.5 rounded-xl border bg-card p-3 transition-shadow hover:shadow-md">
   217→          <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-purple-100 dark:bg-purple-900/30">
   218→            <Clock className="h-4 w-4 text-purple-600 dark:text-purple-400" />
   219→          </div>
   220→          <div className="min-w-0">
   221→            <p className="text-[11px] text-muted-foreground">Avg. Order</p>
   222→            <p className="text-sm font-bold">NPR {npr(749)}</p>
   223→          </div>
   224→        </div>
   225→      </div>
   226→
   227→      <div className="flex flex-col gap-4 lg:flex-row lg:gap-6">
   228→        {/* Left: Products */}
   229→        <div className="flex-1 space-y-4">
   230→          {/* Search + Barcode */}
   231→          <div className="flex gap-2">
   232→            <div className="relative flex-1">
   233→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   234→              <Input
   235→                placeholder="Search products..."
   236→                value={searchQuery}
   237→                onChange={(e) => setSearchQuery(e.target.value)}
   238→                className="pl-9"
   239→              />
   240→            </div>
   241→            <div className="relative w-44 hidden sm:block">
   242→              <Barcode className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   243→              <Input
   244→                placeholder="Scan barcode/PLU"
   245→                value={barcodeInput}
   246→                onChange={(e) => setBarcodeInput(e.target.value)}
   247→                onKeyDown={(e) => e.key === 'Enter' && handleBarcodeScan()}
   248→                className="pl-9"
   249→              />
   250→            </div>
   251→          </div>
   252→
   253→          {/* Category Pills */}
   254→          <ScrollArea className="w-full">
   255→            <div className="flex gap-2 pb-1">
   256→              {categories.map((cat) => (
   257→                <Button
   258→                  key={cat}
   259→                  variant={selectedCategory === cat ? 'default' : 'outline'}
   260→                  size="sm"
   261→                  onClick={() => setSelectedCategory(cat)}
   262→                  className="shrink-0"
   263→                >
   264→                  {cat}
   265→                </Button>
   266→              ))}
   267→            </div>
   268→          </ScrollArea>
   269→
   270→          {/* Product Grid */}
   271→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-2 md:grid-cols-3">
   272→            {filteredProducts.map((product) => {
   273→              const isOutOfStock = product.stock <= 0;
   274→              const isLowStock = product.stock > 0 && product.stock <= product.minStock;
   275→              return (
   276→                <Card
   277→                  key={product.id}
   278→                  className={cn(
   279→                    'cursor-pointer transition-all duration-200 hover:shadow-md hover:-translate-y-0.5',
   280→                    isOutOfStock && 'cursor-not-allowed opacity-50',
   281→                    isLowStock && !isOutOfStock && 'border-amber-400 dark:border-amber-600'
   282→                  )}
   283→                  onClick={() => !isOutOfStock && addToCart(product)}
   284→                >
   285→                  <CardContent className="p-3">
   286→                    <div className="space-y-2">
   287→                      <div className="flex items-start justify-between gap-1">
   288→                        <div className="text-sm font-medium leading-tight min-w-0">
   289→                          {product.name}
   290→                        </div>
   291→                        {isLowStock && !isOutOfStock && (
   292→                          <Badge variant="secondary" className="shrink-0 bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400 text-[10px] px-1.5">
   293→                            Low
   294→                          </Badge>
   295→                        )}
   296→                      </div>
   297→                      <div className="text-lg font-bold">NPR {npr(product.price)}</div>
   298→                      <div className="flex items-center justify-between">
   299→                        <span className="text-xs text-muted-foreground">Stock: {product.stock}</span>
   300→                        {!isOutOfStock ? (
   301→                          <Button size="sm" className="h-7 text-xs gap-1">
   302→                            <Plus className="h-3 w-3" /> Add
   303→                          </Button>
   304→                        ) : (
   305→                          <Badge variant="destructive" className="text-[10px]">Out</Badge>
   306→                        )}
   307→                      </div>
   308→                    </div>
   309→                  </CardContent>
   310→                </Card>
   311→              );
   312→            })}
   313→          </div>
   314→        </div>
   315→
   316→        {/* Right: Cart */}
   317→        <div className="w-full lg:w-[400px]">
   318→          <Card className="sticky top-20">
   319→            <CardHeader className="pb-3">
   320→              <div className="flex items-center justify-between">
   321→                <CardTitle className="flex items-center gap-2 text-base">
   322→                  <ShoppingCart className="h-5 w-5" />
   323→                  Cart
   324→                </CardTitle>
   325→                {cart.length > 0 && (
   326→                  <div className="flex items-center gap-2">
   327→                    <Badge variant="secondary">{cart.reduce((s, i) => s + i.quantity, 0)} items</Badge>
   328→                    <Button variant="ghost" size="icon" className="h-7 w-7 text-muted-foreground hover:text-destructive" onClick={() => { setCart([]); setDiscount(0); }}>
   329→                      <Trash2 className="h-3.5 w-3.5" />
   330→                    </Button>
   331→                  </div>
   332→                )}
   333→              </div>
   334→            </CardHeader>
   335→
   336→            <CardContent className="space-y-4">
   337→              {/* Cart Items */}
   338→              <ScrollArea className="max-h-72">
   339→                {cart.length === 0 ? (
   340→                  <div className="flex h-32 flex-col items-center justify-center gap-2 text-sm text-muted-foreground">
   341→                    <ShoppingCart className="h-8 w-8 opacity-30" />
   342→                    Cart is empty
   343→                  </div>
   344→                ) : (
   345→                  <div className="space-y-2 pr-3">
   346→                    {cart.map((item) => (
   347→                      <div key={item.product.id} className="flex items-center gap-2 rounded-lg bg-muted/50 p-2 transition-colors hover:bg-muted">
   348→                        <div className="min-w-0 flex-1">
   349→                          <div className="text-sm font-medium leading-tight truncate">{item.product.name}</div>
   350→                          <div className="text-xs text-muted-foreground">NPR {npr(item.product.price)} each</div>
   351→                        </div>
   352→                        <div className="flex items-center gap-1 shrink-0">
   353→                          <Button
   354→                            variant="outline"
   355→                            size="icon"
   356→                            className="h-6 w-6"
   357→                            onClick={() => updateQuantity(item.product.id, -1)}
   358→                          >
   359→                            <Minus className="h-3 w-3" />
   360→                          </Button>
   361→                          <span className="w-7 text-center text-sm font-medium">{item.quantity}</span>
   362→                          <Button
   363→                            variant="outline"
   364→                            size="icon"
   365→                            className="h-6 w-6"
   366→                            onClick={() => updateQuantity(item.product.id, 1)}
   367→                          >
   368→                            <Plus className="h-3 w-3" />
   369→                          </Button>
   370→                        </div>
   371→                        <div className="w-20 text-right text-sm font-semibold shrink-0">
   372→                          NPR {npr(item.product.price * item.quantity)}
   373→                        </div>
   374→                        <Button
   375→                          variant="ghost"
   376→                          size="icon"
   377→                          className="h-6 w-6 shrink-0 text-muted-foreground hover:text-destructive"
   378→                          onClick={() => removeFromCart(item.product.id)}
   379→                        >
   380→                          <X className="h-3 w-3" />
   381→                        </Button>
   382→                      </div>
   383→                    ))}
   384→                  </div>
   385→                )}
   386→              </ScrollArea>
   387→
   388→              {/* Totals */}
   389→              {cart.length > 0 && (
   390→                <>
   391→                  <Separator />
   392→                  <div className="space-y-2 text-sm">
   393→                    <div className="flex justify-between">
   394→                      <span className="text-muted-foreground">Subtotal</span>
   395→                      <span>NPR {npr(subtotal)}</span>
   396→                    </div>
   397→                    <div className="flex items-center justify-between gap-2">
   398→                      <span className="text-muted-foreground">Discount</span>
   399→                      <Input
   400→                        type="number"
   401→                        min={0}
   402→                        value={discount || ''}
   403→                        onChange={(e) => setDiscount(Math.max(0, Number(e.target.value)))}
   404→                        placeholder="0"
   405→                        className="h-8 w-28 text-right"
   406→                      />
   407→                    </div>
   408→                    <div className="flex justify-between">
   409→                      <span className="text-muted-foreground">VAT (13%)</span>
   410→                      <span>NPR {npr(Math.round(vat * 100) / 100)}</span>
   411→                    </div>
   412→                    <Separator />
   413→                    <div className="flex justify-between text-lg font-bold">
   414→                      <span>Total</span>
   415→                      <span>NPR {npr(Math.round(total * 100) / 100)}</span>
   416→                    </div>
   417→                  </div>
   418→
   419→                  {/* Payment Methods */}
   420→                  <div className="space-y-2">
   421→                    <div className="text-xs font-medium text-muted-foreground uppercase tracking-wider">Payment Method</div>
   422→                    <div className="grid grid-cols-2 gap-2">
   423→                      {PAYMENT_METHODS.map((method) => {
   424→                        const Icon = method.icon;
   425→                        return (
   426→                          <Button
   427→                            key={method.id}
   428→                            variant={paymentMethod === method.id ? 'default' : 'outline'}
   429→                            className={cn('h-10 justify-start gap-2 transition-all',
   430→                              paymentMethod === method.id && 'shadow-sm'
   431→                            )}
   432→                            onClick={() => setPaymentMethod(method.id)}
   433→                          >
   434→                            <Icon className="h-4 w-4" />
   435→                            {method.label}
   436→                          </Button>
   437→                        );
   438→                      })}
   439→                    </div>
   440→                  </div>
   441→
   442→                  {/* Action Buttons */}
   443→                  <div className="flex gap-2">
   444→                    <Button
   445→                      variant="outline"
   446→                      className="flex-1 gap-2 border-amber-300 text-amber-700 hover:bg-amber-50 dark:border-amber-700 dark:text-amber-400 dark:hover:bg-amber-950/30"
   447→                      onClick={handleHoldSale}
   448→                    >
   449→                      <Pause className="h-4 w-4" />
   450→                      Hold
   451→                    </Button>
   452→                    <Button className="flex-[2] h-12 text-base font-semibold gap-2" onClick={handleCompleteSale}>
   453→                      <Receipt className="h-5 w-5" />
   454→                      Complete Sale
   455→                    </Button>
   456→                  </div>
   457→                </>
   458→              )}
   459→
   460→              {/* Held Sales - Mobile Access */}
   461→              {heldSales.length > 0 && cart.length === 0 && (
   462→                <>
   463→                  <Separator />
   464→                  <Button
   465→                    variant="outline"
   466→                    className="w-full gap-2 border-amber-300 text-amber-700 hover:bg-amber-50 dark:border-amber-700 dark:text-amber-400 dark:hover:bg-amber-950/30"
   467→                    onClick={() => setHeldListOpen(true)}
   468→                  >
   469→                    <PlayCircle className="h-4 w-4" />
   470→                    Resume Held Sale ({heldSales.length})
   471→                  </Button>
   472→                </>
   473→              )}
   474→            </CardContent>
   475→          </Card>
   476→        </div>
   477→      </div>
   478→
   479→      {/* Held Sales Dialog */}
   480→      <Dialog open={heldListOpen} onOpenChange={setHeldListOpen}>
   481→        <DialogContent className="sm:max-w-md">
   482→          <DialogHeader>
   483→            <DialogTitle className="flex items-center gap-2">
   484→              <Pause className="h-5 w-5" />
   485→              Held Sales ({heldSales.length})
   486→            </DialogTitle>
   487→          </DialogHeader>
   488→          <ScrollArea className="max-h-80">
   489→            {heldSales.length === 0 ? (
   490→              <div className="py-8 text-center text-sm text-muted-foreground">
   491→                No held sales
   492→              </div>
   493→            ) : (
   494→              <div className="space-y-2">
   495→                {heldSales.map((held) => (
   496→                  <div key={held.id} className="flex items-center justify-between rounded-lg border p-3 transition-colors hover:bg-muted/50">
   497→                    <div className="min-w-0 flex-1">
   498→                      <div className="text-sm font-medium">{held.cart.reduce((s, i) => s + i.quantity, 0)} items</div>
   499→                      <div className="text-xs text-muted-foreground">
   500→                        {held.cart.map(i => i.product.name).join(', ')}
   501→                      </div>
   502→                      <div className="mt-1 text-xs text-muted-foreground">Held at {formatDateTime(held.heldAt)}</div>
   503→                    </div>
   504→                    <div className="flex items-center gap-2 shrink-0">
   505→                      <span className="text-sm font-bold mr-1">NPR {npr(held.total)}</span>
