   144→/* ------------------------------------------------------------------ */
   145→/*  Helper: Render markdown-like content from store                     */
   146→/* ------------------------------------------------------------------ */
   147→function RenderMarkdown({ content }: { content: string }) {
   148→  const lines = content.split('\n');
   149→  const elements: React.ReactNode[] = [];
   150→  let key = 0;
   151→
   152→  for (const line of lines) {
   153→    const trimmed = line.trim();
   154→    if (!trimmed) {
   155→      elements.push(<div key={key++} className="h-2" />);
   156→      continue;
   157→    }
   158→    if (trimmed.startsWith('### ')) {
   159→      elements.push(<h3 key={key++} className="text-base font-semibold mt-4">{trimmed.slice(4)}</h3>);
   160→    } else if (trimmed.startsWith('## ')) {
   161→      elements.push(<h2 key={key++} className="text-lg font-semibold mt-4">{trimmed.slice(3)}</h2>);
   162→    } else if (trimmed.startsWith('# ')) {
   163→      elements.push(<h1 key={key++} className="text-xl font-bold mt-2">{trimmed.slice(2)}</h1>);
   164→    } else if (trimmed.startsWith('- ')) {
   165→      const boldMatch = trimmed.match(/^- (\*\*(.+?)\*\*:)?(.*)$/);
   166→      const label = boldMatch?.[2];
   167→      const rest = boldMatch?.[3] || trimmed.slice(2);
   168→      elements.push(
   169→        <li key={key++} className="text-sm text-muted-foreground ml-4 list-disc">
   170→          {label && <strong className="text-foreground font-medium">{label}:</strong>}
   171→          <span dangerouslySetInnerHTML={{ __html: rest.replace(/\*\*(.+?)\*\*/g, '<strong class="font-medium text-foreground">$1</strong>') }} />
   172→        </li>
   173→      );
   174→    } else {
   175→      elements.push(
   176→        <p key={key++} className="text-sm text-muted-foreground leading-relaxed" dangerouslySetInnerHTML={{ __html: trimmed.replace(/\*\*(.+?)\*\*/g, '<strong class="font-medium text-foreground">$1</strong>') }} />
   177→      );
   178→    }
   179→  }
   180→
   181→  return <div className="space-y-1">{elements}</div>;
   182→}
