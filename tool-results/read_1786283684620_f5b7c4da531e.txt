    83→
    84→type StatusFilter = 'all' | 'active' | 'inactive';
    85→
    86→const ITEMS_PER_PAGE = 5;
    87→
    88→type TenantSortField = 'name' | 'plan' | 'status' | 'products';
    89→
    90→export default function TenantManagement() {
    91→  const [search, setSearch] = useState('');
    92→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('all');
    93→  const [page, setPage] = useState(0);
    94→  const [addOpen, setAddOpen] = useState(false);
    95→  const [detailOpen, setDetailOpen] = useState(false);
    96→  const [selectedTenant, setSelectedTenant] = useState<Tenant | null>(null);
    97→  const [sortField, setSortField] = useState<TenantSortField>('name');
    98→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc');
    99→
   100→  // Add form state
   101→  const [formName, setFormName] = useState('');
   102→  const [formEmail, setFormEmail] = useState('');
   103→  const [formPhone, setFormPhone] = useState('');
   104→  const [formDomain, setFormDomain] = useState('');
   105→  const [formPlan, setFormPlan] = useState('Plan 1');
   106→
   107→  // Local tenant list for toggle/delete
   108→  const [tenants, setTenants] = useState<Tenant[]>(mockTenants);
   109→
   110→  const filteredTenants = useMemo(() => {
   111→    let result = tenants.filter((t) => {
   112→      const matchesSearch = t.name.toLowerCase().includes(search.toLowerCase());
   113→      const matchesStatus =
   114→        statusFilter === 'all' || t.status === statusFilter;
   115→      return matchesSearch && matchesStatus;
   116→    });
   117→    const planOrder: Record<string, number> = { 'Plan 1': 0, 'Plan 2': 1, 'Plan 3': 2, 'Plan 4': 3 };
   118→    result.sort((a, b) => {
   119→      let cmp = 0;
   120→      if (sortField === 'name') cmp = a.name.localeCompare(b.name);
   121→      else if (sortField === 'plan') cmp = planOrder[a.plan] - planOrder[b.plan];
   122→      else if (sortField === 'status') cmp = a.status.localeCompare(b.status);
   123→      else cmp = a.productCount - b.productCount;
   124→      return sortDir === 'asc' ? cmp : -cmp;
   125→    });
   126→    return result;
   127→  }, [tenants, search, statusFilter, sortField, sortDir]);
   128→
   129→  const totalPages = Math.ceil(filteredTenants.length / ITEMS_PER_PAGE);
   130→  const pagedTenants = filteredTenants.slice(
   131→    page * ITEMS_PER_PAGE,
   132→    (page + 1) * ITEMS_PER_PAGE
   133→  );
   134→
   135→  const handleSort = (field: TenantSortField) => {
   136→    if (sortField === field) setSortDir(d => d === 'asc' ? 'desc' : 'asc');
   137→    else { setSortField(field); setSortDir('asc'); }
   138→  };
   139→  const sortIcon = (field: TenantSortField) => sortField === field ? (sortDir === 'asc' ? '↑' : '↓') : '↕';
   140→
   141→  const handleAddTenant = () => {
   142→    const newTenant: Tenant = {
   143→      id: `t${Date.now()}`,
   144→      name: formName,
   145→      email: formEmail,
   146→      phone: formPhone,
   147→      domain: formDomain,
   148→      plan: formPlan,
   149→      status: 'active',
   150→      createdAt: new Date().toISOString().slice(0, 10),
   151→      ownerName: formName,
   152→      productCount: 0,
   153→      monthlyRevenue: 0,
   154→    };
   155→    setTenants(prev => [newTenant, ...prev]);
   156→    toast.success('Tenant added successfully (mock)');
   157→    setAddOpen(false);
   158→    setFormName('');
   159→    setFormEmail('');
   160→    setFormPhone('');
   161→    setFormDomain('');
   162→    setFormPlan('Plan 1');
   163→  };
   164→
   165→  const handleToggleStatus = (tenant: Tenant) => {
   166→    const newStatus: TenantStatus =
   167→      tenant.status === 'active' ? 'inactive' : 'active';
   168→    setTenants((prev) =>
   169→      prev.map((t) => (t.id === tenant.id ? { ...t, status: newStatus } : t))
   170→    );
   171→    toast.success(`${tenant.name} ${newStatus === 'active' ? 'enabled' : 'disabled'}`);
   172→  };
   173→
   174→  const handleDelete = (tenant: Tenant) => {
   175→    setTenants((prev) => prev.filter((t) => t.id !== tenant.id));
   176→    toast.success(`${tenant.name} deleted (mock)`);
   177→  };
   178→
   179→  const handleViewDetails = (tenant: Tenant) => {
   180→    setSelectedTenant(tenant);
   181→    setDetailOpen(true);
   182→  };
   183→
   184→  return (
   185→    <div className="space-y-6">
   186→      <PageHeader title="Tenant Management">
   187→        <Dialog open={addOpen} onOpenChange={setAddOpen}>
   188→          <DialogTrigger asChild>
   189→            <Button>
   190→              <Plus className="h-4 w-4" />
   191→              Add Tenant
   192→            </Button>
   193→          </DialogTrigger>
   194→          <DialogContent>
   195→            <DialogHeader>
   196→              <DialogTitle>Add New Tenant</DialogTitle>
   197→              <DialogDescription>
   198→                Fill in the details to register a new tenant on the platform.
   199→              </DialogDescription>
   200→            </DialogHeader>
   201→            <div className="grid gap-4 py-2">
   202→              <div className="grid gap-2">
   203→                <Label htmlFor="tenant-name">Name</Label>
   204→                <Input
   205→                  id="tenant-name"
   206→                  placeholder="Store name"
   207→                  value={formName}
   208→                  onChange={(e) => setFormName(e.target.value)}
   209→                />
   210→              </div>
   211→              <div className="grid gap-2">
   212→                <Label htmlFor="tenant-email">Email</Label>
   213→                <Input
   214→                  id="tenant-email"
   215→                  type="email"
   216→                  placeholder="admin@store.com"
   217→                  value={formEmail}
   218→                  onChange={(e) => setFormEmail(e.target.value)}
   219→                />
   220→              </div>
   221→              <div className="grid gap-2">
   222→                <Label htmlFor="tenant-phone">Phone</Label>
   223→                <Input
   224→                  id="tenant-phone"
   225→                  placeholder="+977-98XXXXXXXX"
   226→                  value={formPhone}
   227→                  onChange={(e) => setFormPhone(e.target.value)}
   228→                />
   229→              </div>
   230→              <div className="grid gap-2">
   231→                <Label htmlFor="tenant-domain">Domain</Label>
   232→                <Input
   233→                  id="tenant-domain"
   234→                  placeholder="store.posnepal.com"
   235→                  value={formDomain}
   236→                  onChange={(e) => setFormDomain(e.target.value)}
   237→                />
   238→              </div>
   239→              <div className="grid gap-2">
   240→                <Label htmlFor="tenant-plan">Plan</Label>
   241→                <Select value={formPlan} onValueChange={setFormPlan}>
   242→                  <SelectTrigger className="w-full">
   243→                    <SelectValue />
   244→                  </SelectTrigger>
   245→                  <SelectContent>
   246→                    <SelectItem value="Plan 1">Plan 1</SelectItem>
   247→                    <SelectItem value="Plan 2">Plan 2</SelectItem>
   248→                    <SelectItem value="Plan 3">Plan 3</SelectItem>
   249→                    <SelectItem value="Plan 4">Plan 4</SelectItem>
   250→                  </SelectContent>
   251→                </Select>
   252→              </div>
   253→            </div>
   254→            <DialogFooter>
   255→              <Button variant="outline" onClick={() => setAddOpen(false)}>
   256→                Cancel
   257→              </Button>
   258→              <Button onClick={handleAddTenant}>Add Tenant</Button>
   259→            </DialogFooter>
   260→          </DialogContent>
   261→        </Dialog>
   262→      </PageHeader>
   263→
   264→      {/* Summary Stats */}
   265→      <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   266→        <Card className="transition-shadow hover:shadow-md">
   267→          <CardContent className="p-4">
   268→            <div className="flex items-center justify-between">
   269→              <div>
   270→                <p className="text-xs text-muted-foreground">Total Tenants</p>
   271→                <p className="text-2xl font-semibold">{tenants.length}</p>
   272→              </div>
   273→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-blue-100 dark:bg-blue-900/30 text-blue-700 dark:text-blue-400">
   274→                <Building2 className="h-5 w-5" />
   275→              </div>
   276→            </div>
   277→          </CardContent>
   278→        </Card>
   279→        <Card className="transition-shadow hover:shadow-md">
   280→          <CardContent className="p-4">
   281→            <div className="flex items-center justify-between">
   282→              <div>
   283→                <p className="text-xs text-muted-foreground">Active Tenants</p>
   284→                <p className="text-2xl font-semibold text-emerald-600 dark:text-emerald-400">{tenants.filter(t => t.status === 'active').length}</p>
   285→              </div>
   286→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/30 text-emerald-700 dark:text-emerald-400">
   287→                <CheckCircle2 className="h-5 w-5" />
   288→              </div>
   289→            </div>
   290→          </CardContent>
   291→        </Card>
   292→        <Card className="transition-shadow hover:shadow-md">
   293→          <CardContent className="p-4">
   294→            <div className="flex items-center justify-between">
   295→              <div>
   296→                <p className="text-xs text-muted-foreground">Total Monthly Revenue</p>
   297→                <p className="text-2xl font-semibold">NPR {npr(tenants.reduce((s, t) => s + t.monthlyRevenue, 0))}</p>
   298→              </div>
   299→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-purple-100 dark:bg-purple-900/30 text-purple-700 dark:text-purple-400">
   300→                <CircleDollarSign className="h-5 w-5" />
   301→              </div>
   302→            </div>
   303→          </CardContent>
   304→        </Card>
   305→        <Card className="transition-shadow hover:shadow-md">
   306→          <CardContent className="p-4">
   307→            <div className="flex items-center justify-between">
   308→              <div>
   309→                <p className="text-xs text-muted-foreground">Avg Revenue / Tenant</p>
   310→                <p className="text-2xl font-semibold">NPR {npr(tenants.reduce((s, t) => s + t.monthlyRevenue, 0) / tenants.length)}</p>
   311→              </div>
   312→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/30 text-amber-700 dark:text-amber-400">
   313→                <ArrowUpRight className="h-5 w-5" />
   314→              </div>
   315→            </div>
   316→          </CardContent>
   317→        </Card>
   318→      </div>
   319→
   320→      {/* Search & Filters */}
   321→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   322→        <div className="relative w-full sm:max-w-xs">
   323→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   324→          <Input
   325→            placeholder="Search tenants..."
   326→            className="pl-9"
   327→            value={search}
   328→            onChange={(e) => {
   329→              setSearch(e.target.value);
   330→              setPage(0);
   331→            }}
   332→          />
   333→        </div>
   334→        <div className="flex gap-1">
   335→          {(['all', 'active', 'inactive'] as StatusFilter[]).map((filter) => (
   336→            <Button
   337→              key={filter}
   338→              variant={statusFilter === filter ? 'default' : 'outline'}
   339→              size="sm"
   340→              onClick={() => {
   341→                setStatusFilter(filter);
   342→                setPage(0);
   343→              }}
   344→            >
   345→              {filter.charAt(0).toUpperCase() + filter.slice(1)}
   346→            </Button>
   347→          ))}
   348→        </div>
   349→      </div>
   350→
   351→      {/* Tenant Table */}
   352→      <Card className="transition-shadow hover:shadow-md">
   353→        <CardContent className="p-0">
   354→          <div className="overflow-x-auto">
   355→          <Table>
   356→            <TableHeader>
   357→              <TableRow className="transition-colors hover:bg-muted/50">
   358→                <TableHead className="cursor-pointer select-none" onClick={() => handleSort('name')}>Name <span className="ml-1 text-[10px] opacity-60">{sortIcon('name')}</span></TableHead>
   359→                <TableHead className="hidden md:table-cell">Email</TableHead>
   360→                <TableHead className="cursor-pointer select-none" onClick={() => handleSort('plan')}>Plan <span className="ml-1 text-[10px] opacity-60">{sortIcon('plan')}</span></TableHead>
   361→                <TableHead className="cursor-pointer select-none" onClick={() => handleSort('status')}>Status <span className="ml-1 text-[10px] opacity-60">{sortIcon('status')}</span></TableHead>
   362→                <TableHead className="hidden sm:table-cell cursor-pointer select-none" onClick={() => handleSort('products')}>Products <span className="ml-1 text-[10px] opacity-60">{sortIcon('products')}</span></TableHead>
   363→                <TableHead className="hidden sm:table-cell">Revenue</TableHead>
   364→                <TableHead className="text-right">Actions</TableHead>
   365→              </TableRow>
   366→            </TableHeader>
   367→            <TableBody>
   368→              {pagedTenants.length === 0 ? (
   369→                <TableRow className="transition-colors hover:bg-muted/50">
   370→                  <TableCell colSpan={7} className="h-24 text-center text-muted-foreground">
   371→                    No tenants found.
   372→                  </TableCell>
   373→                </TableRow>
   374→              ) : (
   375→                pagedTenants.map((tenant) => (
   376→                  <TableRow key={tenant.id} className="transition-colors hover:bg-muted/50">
   377→                    <TableCell className="font-medium">{tenant.name}</TableCell>
   378→                    <TableCell className="hidden md:table-cell text-muted-foreground">
   379→                      {tenant.email}
   380→                    </TableCell>
   381→                    <TableCell>
   382→                      <Badge variant="secondary">
   383→                        {tenant.plan.charAt(0).toUpperCase() + tenant.plan.slice(1)}
   384→                      </Badge>
   385→                    </TableCell>
   386→                    <TableCell>
   387→                      <Badge
   388→                        className={getStatusBadgeClasses(tenant.status)}
   389→                      >
   390→                        {tenant.status.charAt(0).toUpperCase() + tenant.status.slice(1)}
   391→                      </Badge>
   392→                    </TableCell>
   393→                    <TableCell className="hidden sm:table-cell">
   394→                      {tenant.productCount}
   395→                    </TableCell>
   396→                    <TableCell className="hidden sm:table-cell">
   397→                      NPR {npr(tenant.monthlyRevenue)}
   398→                    </TableCell>
   399→                    <TableCell className="text-right">
   400→                      <div className="flex items-center justify-end gap-1">
   401→                        <Button
   402→                          variant="ghost"
   403→                          size="icon"
   404→                          onClick={() => handleViewDetails(tenant)}
   405→                        >
   406→                          <Eye className="h-4 w-4" />
   407→                        </Button>
   408→                        <Button
   409→                          variant="ghost"
   410→                          size="icon"
   411→                          onClick={() => handleToggleStatus(tenant)}
   412→                        >
   413→                          <ToggleLeft className="h-4 w-4" />
   414→                        </Button>
   415→                        <Button
   416→                          variant="ghost"
   417→                          size="icon"
   418→                          className="text-destructive hover:text-destructive"
   419→                          onClick={() => handleDelete(tenant)}
   420→                        >
   421→                          <Trash2 className="h-4 w-4" />
   422→                        </Button>
   423→                      </div>
   424→                    </TableCell>
   425→                  </TableRow>
   426→                ))
   427→              )}
   428→            </TableBody>
   429→          </Table>
   430→          </div>
   431→        </CardContent>
   432→      </Card>
   433→
   434→      {/* Pagination */}
   435→      {totalPages > 1 && (
   436→        <div className="flex items-center justify-between">
   437→          <p className="text-sm text-muted-foreground">
   438→            Showing {page * ITEMS_PER_PAGE + 1}–{Math.min((page + 1) * ITEMS_PER_PAGE, filteredTenants.length)} of{' '}
   439→            {filteredTenants.length}
   440→          </p>
   441→          <div className="flex items-center gap-2">
   442→            <Button
   443→              variant="outline"
   444→              size="icon"
   445→              disabled={page === 0}
   446→              onClick={() => setPage(page - 1)}
   447→            >
   448→              <ChevronLeft className="h-4 w-4" />
   449→            </Button>
   450→            {Array.from({ length: totalPages }, (_, i) => (
   451→              <Button
   452→                key={i}
   453→                variant={page === i ? 'default' : 'outline'}
   454→                size="icon"
   455→                className="h-9 w-9"
   456→                onClick={() => setPage(i)}
   457→              >
   458→                {i + 1}
   459→              </Button>
   460→            ))}
   461→            <Button
   462→              variant="outline"
   463→              size="icon"
   464→              disabled={page >= totalPages - 1}
   465→              onClick={() => setPage(page + 1)}
   466→            >
   467→              <ChevronRight className="h-4 w-4" />
   468→            </Button>
   469→          </div>
   470→        </div>
   471→      )}
   472→
   473→      {/* View Details Dialog */}
   474→      <Dialog open={detailOpen} onOpenChange={setDetailOpen}>
   475→        <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   476→          <DialogHeader>
   477→            <DialogTitle className="flex items-center gap-2">
   478→              <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10 text-primary">
   479→                <Building2 className="h-4 w-4" />
   480→              </div>
   481→              Tenant Details
   482→            </DialogTitle>
   483→            <DialogDescription>Full information and recent activity.</DialogDescription>
   484→          </DialogHeader>
   485→          {selectedTenant && (
   486→            <div className="grid gap-5 py-2">
   487→              {/* Tenant Header Card */}
   488→              <div className={"rounded-lg border p-4 " + planColors[selectedTenant.plan].bg + ' ' + planColors[selectedTenant.plan].border}>
   489→                <div className="flex items-start justify-between">
   490→                  <div className="flex items-center gap-3">
   491→                    <div className={"flex h-12 w-12 items-center justify-center rounded-xl text-lg font-bold text-white bg-primary">
   492→                      {selectedTenant.name.charAt(0)}
   493→                    </div>
   494→                    <div>
   495→                      <p className="font-semibold text-base">{selectedTenant.name}</p>
   496→                      <p className="text-xs text-muted-foreground">Owned by {selectedTenant.ownerName}</p>
   497→                      <div className="mt-1 flex items-center gap-2">
   498→                        <Badge className={getPlanBadgeClasses(selectedTenant.plan)} variant="secondary">
   499→                          {selectedTenant.plan.charAt(0).toUpperCase() + selectedTenant.plan.slice(1)} Plan
   500→                        </Badge>
   501→                        <Badge className={getStatusBadgeClasses(selectedTenant.status)}>
   502→                          {selectedTenant.status.charAt(0).toUpperCase() + selectedTenant.status.slice(1)}
   503→                        </Badge>
   504→                      </div>
   505→                    </div>
   506→                  </div>
   507→                  {selectedTenant.plan === 'Plan 4' && (
   508→                    <Crown className="h-5 w-5 text-amber-500" />
   509→                  )}
   510→                </div>
   511→              </div>
   512→
   513→              {/* Contact Info Grid */}
   514→              <div className="grid grid-cols-2 gap-3">
   515→                <div className="rounded-lg border p-3">
   516→                  <div className="flex items-center gap-2 text-xs text-muted-foreground">
