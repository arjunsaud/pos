   142→/* ------------------------------------------------------------------ */
   143→/*  Helper: Render markdown-like content from store                     */
   144→/* ------------------------------------------------------------------ */
   145→function RenderMarkdown({ content }: { content: string }) {
   146→  const lines = content.split('\n');
   147→  const elements: React.ReactNode[] = [];
   148→  let key = 0;
   149→
   150→  for (const line of lines) {
   151→    const trimmed = line.trim();
   152→    if (!trimmed) {
   153→      elements.push(<div key={key++} className="h-2" />);
   154→      continue;
   155→    }
   156→    if (trimmed.startsWith('### ')) {
   157→      elements.push(<h3 key={key++} className="text-base font-semibold mt-4">{trimmed.slice(4)}</h3>);
   158→    } else if (trimmed.startsWith('## ')) {
   159→      elements.push(<h2 key={key++} className="text-lg font-semibold mt-4">{trimmed.slice(3)}</h2>);
   160→    } else if (trimmed.startsWith('# ')) {
   161→      elements.push(<h1 key={key++} className="text-xl font-bold mt-2">{trimmed.slice(2)}</h1>);
   162→    } else if (trimmed.startsWith('- ')) {
   163→      const boldMatch = trimmed.match(/^- (\*\*(.+?)\*\*:)?(.*)$/);
   164→      const label = boldMatch?.[2];
   165→      const rest = boldMatch?.[3] || trimmed.slice(2);
   166→      elements.push(
   167→        <li key={key++} className="text-sm text-muted-foreground ml-4 list-disc">
   168→          {label && <strong className="text-foreground font-medium">{label}:</strong>}
   169→          <span dangerouslySetInnerHTML={{ __html: rest.replace(/\*\*(.+?)\*\*/g, '<strong class="font-medium text-foreground">$1</strong>') }} />
   170→        </li>
   171→      );
   172→    } else {
   173→      elements.push(
   174→        <p key={key++} className="text-sm text-muted-foreground leading-relaxed" dangerouslySetInnerHTML={{ __html: trimmed.replace(/\*\*(.+?)\*\*/g, '<strong class="font-medium text-foreground">$1</strong>') }} />
   175→      );
   176→    }
   177→  }
   178→
   179→  return <div className="space-y-1">{elements}</div>;
   180→}
